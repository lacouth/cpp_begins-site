    Atividade de revisão — Mini-sistema da Biblioteca¶

    Este é o código da atividade. As instruções completas (o que cada classe precisa ter e a saída esperada de cada parte) estão na folha impressa.

    Como começar

    1. Clique no ícone de copiar no canto do bloco abaixo e cole tudo no seu compilador online.
    2. Rode uma vez: deve imprimir só === Biblioteca do Campus ===.
    3. Escreva as classes na área ESCREVA SUAS CLASSES AQUI.
    4. A cada parte pronta, apague as duas linhas de marcação dela no main() (/* PARTE N e PARTE N */), rode e compare com a folha.

    Na saída, 1 quer dizer sim/verdadeiro e 0 quer dizer não/falso.

    // Mini-sistema da Biblioteca — atividade de revisão
//
// O main() abaixo já está pronto: ele é o "sistema" que usa as suas classes.
// Sua tarefa é escrever as classes Livro e Biblioteca na área marcada,
// seguindo a folha impressa.
//
// Como trabalhar:
//   1. Escreva o que a Parte 1 pede.
//   2. No main(), apague as duas linhas de marcação da PARTE 1
//      (a que abre com /* e a que fecha com */).
//   3. Rode e compare a saída com a da folha.
//   4. Repita para as partes 2, 3, 4 e 5.

#include <iostream>
#include <string>
#include <vector>
#include <map>

using std::cout;
using std::string;
using std::vector;
using std::map;

// ============================================================
// ESCREVA SUAS CLASSES AQUI
// ============================================================

// class Livro { ... };

// class Biblioteca { ... };

// ============================================================
// O SISTEMA — não altere nada abaixo, só apague as marcações
// ============================================================

int main() {
    cout << "=== Biblioteca do Campus ===\n";

/* PARTE 1
    cout << "\n--- Parte 1: criando livros ---\n";
    Livro a("Dom Casmurro", "Machado de Assis", 1899);
    Livro b("Memorias Postumas", "Machado de Assis", 1881);
    Livro c("O Cortico", "Aluisio Azevedo", -5);

    cout << a.titulo() << " (" << a.autor() << ", " << a.ano() << ")\n";
    cout << b.titulo() << " (" << b.autor() << ", " << b.ano() << ")\n";
    cout << c.titulo() << " (" << c.autor() << ", " << c.ano() << ")\n";
    cout << "Dom Casmurro disponivel? " << a.disponivel() << "\n";
PARTE 1 */

/* PARTE 2
    cout << "\n--- Parte 2: regras do livro ---\n";
    c.define_ano(1890);
    cout << "Ano corrigido: " << c.ano() << "\n";
    c.define_ano(-1);
    cout << "Depois de tentar ano -1: " << c.ano() << "\n";

    cout << "Emprestar Dom Casmurro: " << a.emprestar() << "\n";
    cout << "Emprestar de novo: " << a.emprestar() << "\n";
    cout << "Disponivel? " << a.disponivel() << "\n";
    a.devolver();
    cout << "Depois de devolver, disponivel? " << a.disponivel() << "\n";
PARTE 2 */

/* PARTE 3
    cout << "\n--- Parte 3: montando o acervo ---\n";
    Biblioteca bib("Biblioteca do Campus");
    bib.adicionar(a);
    bib.adicionar(b);
    bib.adicionar(c);
    bib.adicionar(Livro("Vidas Secas", "Graciliano Ramos", 1938));

    cout << bib.nome() << " tem " << bib.quantidade() << " livros\n";
    bib.listar();
PARTE 3 */

/* PARTE 4
    cout << "\n--- Parte 4: emprestimos ---\n";
    cout << "Emprestar Vidas Secas: " << bib.emprestar("Vidas Secas") << "\n";
    cout << "Emprestar Vidas Secas de novo: " << bib.emprestar("Vidas Secas") << "\n";
    cout << "Emprestar Iracema (nao existe): " << bib.emprestar("Iracema") << "\n";
    cout << "Emprestar Dom Casmurro: " << bib.emprestar("Dom Casmurro") << "\n";
    cout << "Disponiveis: " << bib.disponiveis() << " de " << bib.quantidade() << "\n";

    cout << "Devolver Vidas Secas: " << bib.devolver("Vidas Secas") << "\n";
    cout << "Devolver Vidas Secas de novo: " << bib.devolver("Vidas Secas") << "\n";
    cout << "Disponiveis: " << bib.disponiveis() << " de " << bib.quantidade() << "\n";
    bib.listar();
PARTE 4 */

/* PARTE 5
    cout << "\n--- Parte 5: relatorios ---\n";
    bib.emprestar("Vidas Secas");
    bib.devolver("Vidas Secas");
    bib.emprestar("Vidas Secas");

    cout << "Vidas Secas foi emprestado " << bib.vezes_emprestado("Vidas Secas") << " vezes\n";
    cout << "Dom Casmurro foi emprestado " << bib.vezes_emprestado("Dom Casmurro") << " vezes\n";
    cout << "O Cortico foi emprestado " << bib.vezes_emprestado("O Cortico") << " vezes\n";
    cout << "Mais emprestado: " << bib.mais_emprestado() << "\n";
    cout << "Livros de Machado de Assis: " << bib.livros_do_autor("Machado de Assis") << "\n";
    cout << "Livros de Jorge Amado: " << bib.livros_do_autor("Jorge Amado") << "\n";

    Biblioteca vazia("Sala de Leitura");
    cout << "Mais emprestado na " << vazia.nome() << ": [" << vazia.mais_emprestado() << "]\n";
PARTE 5 */

    return 0;
}
