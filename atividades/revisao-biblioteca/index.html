    Atividade de revisão — Sistema da Biblioteca¶

    Este é o código da atividade. O que cada classe e cada função precisa fazer está na folha impressa.

    Como montar o projeto

    O sistema tem quatro arquivos. No OnlineGDB, o main.cpp já existe: cole nele o conteúdo do main.cpp abaixo. Depois, use o botão de novo arquivo (ícone de folha com +, ao lado da aba main.cpp) para criar livro.h, biblioteca.h e menu.h, com esses nomes exatos, em letras minúsculas, e cole o conteúdo de cada um.

    O programa só compila depois que a Parte 1 (a classe Livro) estiver escrita. A partir daí, cada parte que você completa faz mais opções do menu funcionarem. Os trechos marcados como EXTRA são opcionais: só depois das Partes 1 a 3.

    #ifndef MENU_H
#define MENU_H

#include <iostream>
#include <string>
#include "biblioteca.h"

using std::cin;
using std::cout;
using std::getline;
using std::string;
using std::stoi;

// ============================================================
// Leitura do teclado — já pronta, só use
// ============================================================

string lerTexto(string pergunta) {
    cout << pergunta;
    string texto;
    getline(cin, texto);
    return texto;
}

// Devolve -1 se o que foi digitado não é um número (ou se ficou em branco).
int lerInteiro(string pergunta) {
    string texto = lerTexto(pergunta);
    if (texto.empty()) {
        return -1;
    }
    for (char c : texto) {
        if (c < '0' || c > '9') {
            return -1;
        }
    }
    return stoi(texto);
}

// ============================================================
// Opções do menu
// ============================================================
// Repare que toda função recebe Biblioteca& (por referência):
// sem o &, ela mexeria numa CÓPIA e o main() nunca veria a mudança.

void menuCadastrar(Biblioteca& biblioteca) {
    string titulo = lerTexto("Titulo: ");
    string autor = lerTexto("Autor: ");
    int ano = lerInteiro("Ano: ");

    Livro livro(titulo, autor, ano);
    biblioteca.cadastrar(livro);
    cout << "Livro cadastrado.\n";
}

void menuListar(Biblioteca& biblioteca) {
    biblioteca.listar();
}

void menuEditarAno(Biblioteca& biblioteca) {
    string titulo = lerTexto("Titulo: ");
    int ano = lerInteiro("Novo ano: ");
    if (biblioteca.editarAno(titulo, ano)) {
        cout << "Livro atualizado.\n";
    } else {
        cout << "Livro nao encontrado.\n";
    }
}

void menuRelatorio(Biblioteca& biblioteca) {
    biblioteca.relatorio();
    string titulo = biblioteca.maisEmprestado();
    if (titulo != "") {
        cout << "Mais emprestado: " << titulo << "\n";
    }
}

void menuDevolver(Biblioteca& biblioteca) {
    string titulo = lerTexto("Titulo: ");
    if (biblioteca.devolver(titulo)) {
        cout << "Devolucao registrada.\n";
    } else {
        cout << "Nao foi possivel devolver.\n";
    }
}

    main.cpp — pronto, não precisa alterar
    // Sistema da Biblioteca — atividade de revisão
// Este arquivo já está pronto: não precisa alterar nada aqui.

#include <iostream>
#include "livro.h"
#include "biblioteca.h"
#include "menu.h"

using std::cin;
using std::cout;

int main() {
    Biblioteca biblioteca("Biblioteca do Campus");

    // Acervo inicial, para o sistema não começar vazio.
    biblioteca.cadastrar(Livro("Dom Casmurro", "Machado de Assis", 1899));
    biblioteca.cadastrar(Livro("Memorias Postumas", "Machado de Assis", 1881));
    biblioteca.cadastrar(Livro("Vidas Secas", "Graciliano Ramos", 1938));

    int opcao = -1;
    while (opcao != 0 && cin) {
        cout << "\n[1] Cadastrar  [2] Listar  [3] Emprestar  [4] Editar ano"
             << "  [5] Relatorio  [6] Devolver (extra)  [0] Sair\n";
        opcao = lerInteiro("Opcao: ");

        switch (opcao) {
            case 1: menuCadastrar(biblioteca); break;
            case 2: menuListar(biblioteca); break;
            case 3: menuEmprestar(biblioteca); break;
            case 4: menuEditarAno(biblioteca); break;
            case 5: menuRelatorio(biblioteca); break;
            case 6: menuDevolver(biblioteca); break;
            case 0: cout << "Ate logo!\n"; break;
            default: cout << "Opcao invalida.\n"; break;
        }
    }
    return 0;
}
